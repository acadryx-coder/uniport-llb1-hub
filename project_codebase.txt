export default function ThemeToggle() {
  const [theme, setTheme] = useState(() => document.documentElement.getAttribute('data-theme') || 'light')

  useEffect(() => {
    document.documentElement.setAttribute('data-theme', theme)
    try { localStorage.setItem('llb1.theme', theme) } catch {}
    const meta = document.querySelector('meta[name="theme-color"]')
    if (meta) meta.setAttribute('content', theme === 'dark' ? '#14120f' : '#f7f5f0')
  }, [theme])

  return (
    <button
      className="theme-btn-icon"
      onClick={() => setTheme((t) => (t === 'dark' ? 'light' : 'dark'))}
      aria-label={`Switch to ${theme === 'dark' ? 'light' : 'dark'} mode`}
    >
      {theme === 'dark' ? ICON_MOON : ICON_SUN}
    </button>
  )

import { Navigate, Outlet, useLocation } from 'react-router-dom'
import { useAuth, setPendingRedirect } from '../lib/auth.js'

export default function ProtectedRoute() {
  const { user, loading } = useAuth()
  const location = useLocation()

  // 1. Wait for Supabase to finish checking the session
  if (loading) {
    return (
      <div className="view" style={{ display: 'flex', alignItems: 'center', justifyContent: 'center', height: '100vh', opacity: 0.5 }}>
        <div className="empty">Loading LLB 1 Hub…</div>
      </div>
    )
  }

  // 2. If no user after loading, redirect to login
  if (!user) {
    const target = location.pathname + location.search
    setPendingRedirect(target)
    return <Navigate to="/login" replace state={{ from: target }} />
  }

  // 3. Otherwise, render the app
  return <Outlet />
}

import { useState, useEffect } from 'react'
import { Link } from 'react-router-dom'
import { supabase } from '../lib/supabase.js'
import { useAuth } from '../lib/auth.js'

export default function QuizModal({ isOpen, onClose, quiz, resourceId }) {
  const { user } = useAuth()
  
  const [pastScores, setPastScores] = useState([])
  const [sessionScores, setSessionScores] = useState([])
  const [loadingScores, setLoadingScores] = useState(false)
  
  const [step, setStep] = useState('intro') // 'intro' | 'taking' | 'results'
  const [currentQ, setCurrentQ] = useState(0)
  const [answers, setAnswers] = useState({})
  const [finalScore, setFinalScore] = useState(0)

  const activeQuiz = Array.isArray(quiz) ? quiz[0] : quiz

  // Load past scores from Supabase on open
  useEffect(() => {
    if (isOpen && activeQuiz && user) {
      setLoadingScores(true)
      supabase
        .from('quiz_attempts')
        .select('*')
        .eq('resource_id', resourceId)
        .order('created_at', { ascending: false })
        .limit(5)
        .then(({ data }) => {
          if (data) setPastScores(data)
          setLoadingScores(false)
        })
    }
    
    if (isOpen) {
      setStep('intro')
      setCurrentQ(0)
      setAnswers({})
      setFinalScore(0)
    }
  }, [isOpen, activeQuiz, resourceId, user])

  useEffect(() => {
    const handleEsc = (e) => { if (e.key === 'Escape') onClose() }
    if (isOpen) window.addEventListener('keydown', handleEsc)
    return () => window.removeEventListener('keydown', handleEsc)
  }, [isOpen, onClose])

  if (!isOpen || !activeQuiz || !activeQuiz.questions) return null

  const displayScores = user ? pastScores : sessionScores

  const handleStart = () => {
    setStep('taking')
    setCurrentQ(0)
    setAnswers({})
  }

  const handleSelect = (optIndex) => {
    setAnswers({ ...answers, [currentQ]: optIndex })
  }

  const handleNext = () => {
    if (currentQ < activeQuiz.questions.length - 1) {
      setCurrentQ(currentQ + 1)
    } else {
      submitQuiz()
    }
  }

  const submitQuiz = async () => {
    let correct = 0
    activeQuiz.questions.forEach((q, i) => {
      if (answers[i] === q.answer) correct++
    })
    
    const pct = Math.round((correct / activeQuiz.questions.length) * 100)
    setFinalScore(pct)
    
    const newScore = {
      score: pct,
      correct,
      total: activeQuiz.questions.length,
      created_at: new Date().toISOString()
    }

    if (user) {
      // Save to Supabase
      const { error } = await supabase.from('quiz_attempts').insert([{
        user_id: user.id,
        resource_id: resourceId,
        score: pct,
        total: activeQuiz.questions.length
      }])
      
      if (!error) {
        // Refresh local list of scores
        setPastScores([newScore, ...pastScores].slice(0, 5))
      } else {
        console.error("Failed to save score to DB:", error)
      }
    } else {
      // Save to session state for guests
      setSessionScores([newScore, ...sessionScores].slice(0, 5))
    }
    
    setStep('results')
  }

  const q = activeQuiz.questions[currentQ]
  const isAnswered = answers[currentQ] !== undefined
  const attemptNumber = displayScores.length + 1

  return (
    <div className="modal-backdrop" onClick={onClose}>
      <div className="modal-content quiz-modal" onClick={(e) => e.stopPropagation()}>
        
        {step === 'intro' && (
          <div className="quiz-intro">
            <div className="quiz-header">
              <h2>{activeQuiz.title}</h2>
              <button className="icon-btn" onClick={onClose}>×</button>
            </div>
            <p className="quiz-desc">{activeQuiz.description}</p>
            
            <div className="quiz-note">
              <strong>Note:</strong> This is a self-assessment tool.
              {!user && ' Since you are not logged in, these scores will only last until you close this page.'}
            </div>

            {loadingScores ? (
              <div className="mute-2" style={{ marginTop: 12 }}>Loading scores…</div>
            ) : displayScores.length > 0 && (
              <div className="quiz-history">
                <h4>Recent Scores {!user && '(This Session)'}</h4>
                {displayScores.map((s, i) => (
                  <div key={i} className="quiz-history-row">
                    <span>{new Date(s.created_at).toLocaleDateString('en-NG', { day: 'numeric', month: 'short' })}</span>
                    <span className="quiz-history-score">{s.correct}/{s.total} ({s.score}%)</span>
                  </div>
                ))}
              </div>
            )}

            <button className="btn btn-gold" style={{ width: '100%', marginTop: 20 }} onClick={handleStart}>
              Start {attemptNumber === 1 ? '1st' : attemptNumber === 2 ? '2nd' : attemptNumber === 3 ? '3rd' : `${attemptNumber}th`} attempt
            </button>

            {!user && (
              <div className="quiz-login-prompt">
                Want to save your scores and track your progress? <Link to="/login">Log in here</Link>
              </div>
            )}
          </div>
        )}

        {step === 'taking' && (
          <div className="quiz-taking">
            <div className="quiz-header">
              <div className="quiz-progress-text">Question {currentQ + 1} of {activeQuiz.questions.length}</div>
              <button className="icon-btn" onClick={onClose}>×</button>
            </div>
            
            <div className="quiz-progress-bar">
              <i style={{ width: `${((currentQ + 1) / activeQuiz.questions.length) * 100}%` }} />
            </div>

            <h3 className="quiz-q">{q.q}</h3>
            
            <div className="quiz-options">
              {q.options.map((opt, i) => (
                <button 
                  key={i} 
                  className={`quiz-opt ${answers[currentQ] === i ? 'selected' : ''}`}

function FloatingLogin({ onClick }) {
  const [pos, setPos] = useState({ x: 16, y: 80 })
  const [dragging, setDragging] = useState(false)
  const [rel, setRel] = useState({ x: 0, y: 0 })

  const onPointerDown = (e) => {
    setDragging(true)
    setRel({ x: e.clientX - pos.x, y: e.clientY - pos.y })
    e.target.setPointerCapture(e.pointerId)
  }
  const onPointerMove = (e) => {
    if (!dragging) return
    setPos({ x: e.clientX - rel.x, y: e.clientY - rel.y })
  }
  const onPointerUp = (e) => {
    setDragging(false)
    e.target.releasePointerCapture(e.pointerId)
  }

  return (
    <div
      onPointerDown={onPointerDown}
      onPointerMove={onPointerMove}
      onPointerUp={onPointerUp}
      onClick={onClick}
      style={{
        position: 'absolute', left: pos.x, top: pos.y,
        width: 44, height: 44, borderRadius: '50%',
        background: 'var(--acc)', color: '#fff',
        display: 'flex', alignItems: 'center', justifyContent: 'center',
        boxShadow: '0 4px 12px rgba(0,0,0,0.3)', cursor: dragging ? 'grabbing' : 'grab',

export default function MootTimer() {
  useDocumentTitle('Moot Timer')
  
  const PRESETS = [
    { label: 'Appellant (15m)', seconds: 15 * 60 },
    { label: 'Respondent (15m)', seconds: 15 * 60 },
    { label: 'Rebuttal (5m)', seconds: 5 * 60 },
    { label: 'Practice (10m)', seconds: 10 * 60 },
  ]

  const [timeLeft, setTimeLeft] = useState(15 * 60)
  const [isRunning, setIsRunning] = useState(false)
  const [totalTime, setTotalTime] = useState(15 * 60)
  const timerRef = useRef(null)

  useEffect(() => {
    if (isRunning && timeLeft > 0) {
      timerRef.current = setInterval(() => {
        setTimeLeft(prev => {
          if (prev <= 1) {
            clearInterval(timerRef.current)
            setIsRunning(false)
            return 0
          }
          return prev - 1
        })
      }, 1000)
    }
    return () => clearInterval(timerRef.current)
  }, [isRunning])

  const toggleTimer = () => {
    if (timeLeft === 0) {
      setTimeLeft(totalTime)
      setIsRunning(true)
    } else {
      setIsRunning(!isRunning)
    }
  }
  
  const resetTimer = (seconds = totalTime) => {
    setIsRunning(false)
    setTimeLeft(seconds)
    setTotalTime(seconds)
  }

  const formatTime = (secs) => {
    const m = Math.floor(secs / 60)
    const s = secs % 60
    return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
  }

  const pct = (timeLeft / totalTime) * 100
  const isUrgent = timeLeft <= 60 && timeLeft > 0 // Last minute

function escapeHtml(s) {
  return String(s == null ? '' : s).replace(/[&<>"']/g, (c) => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  }[c]))
}

function slugify(text) {
  return text.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '')
}

function inline(text) {
  let s = escapeHtml(text)
  // Catch long ALL CAPS blocks and wrap them in a span for styling
  s = s.replace(/([A-Z0-9\s,.'"-]{50,})/g, '<span class="statute-callout-block">$1</span>')
  
  s = s.replace(/`([^`\n]+)`/g, '<code>$1</code>')
  s = s.replace(/==([^=\n]+)==/g, '<mark>$1</mark>')
  s = s.replace(/\*\*([^*\n]+)\*\*/g, '<strong>$1</strong>')
  s = s.replace(/(^|[^*\w])\*([^*\n]+)\*/g, '$1<em>$2</em>')
  s = s.replace(
    /\[\[([^\]\n]+)\]\]|\[\^([a-zA-Z0-9]+)\]|\[([^\]\n]+)\]\((https?:\/\/[^\s)]+)\)/g,
    (m, xref, fnref, ltext, lurl) => {
      if (xref !== undefined) return `<a class="xref" href="#" data-xref="${xref}">${xref}</a>`
      if (fnref !== undefined) return `<sup class="fnref" id="fnref-${fnref}"><a href="#fn-${fnref}">${fnref}</a></sup>`
      return `<a href="${lurl}" target="_blank" rel="noopener noreferrer">${ltext}</a>`
    }
  )
  return s
}

export function parseLegalMD(md) {
  if (!md || typeof md !== 'string') return ''
  const footnotes = {}
  
  let text = md.replace(/\r\n/g, '\n')
  // FORCE blockquotes to be on their own lines so they parse correctly
  text = text.replace(/\n>\s/g, '\n\n> ')
  text = text.replace(/^>\s/g, '\n\n> ')

  text = text.replace(/^\[\^([a-zA-Z0-9]+)\]:\s*(.+)$/gm, (m, id, ft) => {
    footnotes[id] = ft.trim()
    return ''
  })

  const blocks = text.split(/\n{2,}/)
  const out = []
  for (const block of blocks) {
    const lines = block.split('\n').filter((l) => l.trim().length)
    if (!lines.length) continue
    for (const rule of blockRules) {
      if (rule.test(lines)) {
        out.push(rule.render(lines))
        break
      }
    }
  }

  const fnKeys = Object.keys(footnotes).sort((a, b) => a.localeCompare(b, undefined, { numeric: true }))
  if (fnKeys.length) {
    out.push(
      '<section class="footnotes"><h4>Notes</h4><ol>' +
        fnKeys.map((k) => `<li id="fn-${k}">${inline(footnotes[k])} <a href="#fnref-${k}" class="fn-back" aria-label="Back to text">↩</a></li>`).join('') +
      '</ol></section>'
    )
  }
  return out.join('\n')
}

export function useLocalStorage(key, initial) {
  const [value, setValue] = useState(() => {
    try {
      const raw = localStorage.getItem(key)
      return raw === null ? initial : JSON.parse(raw)
    } catch {
      return initial
    }
  })

  useEffect(() => {
    try {
      localStorage.setItem(key, JSON.stringify(value))
    } catch {}
  }, [key, value])

  // Listen for local changes from other components in the same tab
  useEffect(() => {
    const handler = (e) => {
      if (e.detail && e.detail.key === key) {
        setValue(e.detail.value)
      }
    }
    window.addEventListener('local-storage', handler)
    return () => window.removeEventListener('local-storage', handler)
  }, [key])

  const setSyncedValue = useCallback((newValue) => {
    setValue((prev) => {
      const next = typeof newValue === 'function' ? newValue(prev) : newValue
      window.dispatchEvent(new CustomEvent('local-storage', { detail: { key, value: next } }))
      return next
    })
  }, [key])

  return [value, setSyncedValue]
}

export function useAuth() {
  const [user, setUser] = useState(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      if (session) fetchProfile(session.user)
      else setLoading(false)
    })
    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
      if (session) fetchProfile(session.user)
      else { setUser(null); setLoading(false) }
    })
    return () => subscription.unsubscribe()
  }, [])

  const fetchProfile = async (authUser) => {
    const { data } = await supabase
      .from('profiles')
      .select('role, full_name, avatar_url, matric_number, level, phone, bio')
      .eq('id', authUser.id)
      .single()
    setUser({
      email: authUser.email,
      role: data?.role || 'student',
      name: data?.full_name,
      avatar: data?.avatar_url,
      matric: data?.matric_number,
      level: data?.level,
      phone: data?.phone,
      bio: data?.bio
    })
    setLoading(false)
  }

  const updateProfile = async (updates) => {
    const { data: { user: authUser } } = await supabase.auth.getUser()
    const { error } = await supabase
      .from('profiles')
      .update(updates)
      .eq('id', authUser.id)
    if (error) throw error
    setUser(prev => ({ ...prev, ...updates }))
  }

  const uploadAvatar = async (file) => {
    const { data: { user: authUser } } = await supabase.auth.getUser()
    const fileExt = file.name.split('.').pop()
    const filePath = `${authUser.id}/avatar_${Date.now()}.${fileExt}`

    const { error: uploadError } = await supabase.storage
      .from('avatars')
      .upload(filePath, file, { cacheControl: '3600', upsert: true })
    if (uploadError) throw uploadError

    const { data: { publicUrl } } = supabase.storage
      .from('avatars')
      .getPublicUrl(filePath)

    await updateProfile({ avatar_url: publicUrl })
    return publicUrl
  }

  return { user, loading, signIn, signOut, updateProfile, uploadAvatar }
}

// Helper that respects the active status of classes
export function getNextClass(timetable, settings) {
  if (!timetable || !timetable.length) return null
  if (settings && !settings.lectures_active) return null // Return nothing if suspended

  const now = new Date()
  const today = now.getDay()
  const nowMin = now.getHours() * 60 + now.getMinutes()
  const toMin = (t) => { const [h, m] = t.split(':'); return parseInt(h) * 60 + parseInt(m) }

  for (let off = 0; off < 8; off++) {
    const day = (today + off) % 7
    const list = timetable.filter((e) => e.day === day && e.active)
    for (const ev of list) {
      const evMin = toMin(ev.start_time)
      if (off === 0 && evMin <= nowMin) continue
      return { event: ev, offset: off, mins: off === 0 ? evMin - nowMin : null }
    }
  }
  return null
}

export function getNextClass() {
  const now = new Date()
  const today = now.getDay()
  const nowMin = now.getHours() * 60 + now.getMinutes()
  for (let off = 0; off < 8; off++) {
    const day = (today + off) % 7
    const list = timetable
      .filter((e) => e.day === day)
      .sort((a, b) => {
        const am = a.start.split(':').reduce((h, m, i) => i === 0 ? +h * 60 : h + +m, 0)
        const bm = b.start.split(':').reduce((h, m, i) => i === 0 ? +h * 60 : h + +m, 0)
        return am - bm
      })
    for (const ev of list) {
      const evMin = ev.start.split(':').map(Number).reduce((h, m, i) => i === 0 ? h * 60 : h + m, 0)
      if (off === 0 && evMin <= nowMin) continue
      return { event: ev, offset: off, mins: off === 0 ? evMin - nowMin : null }
    }
  }
  return null
}

function esc(s) {
  return String(s || '').replace(/[&<>"']/g, (c) => ({
    '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
  }[c]))
}

async function resolveMeta(path, origin) {
  const img = `${origin}/og-default.png`
  const fallback = {
    title: 'LLB 1 Hub · UNIPORT',
    description: 'The official academic hub for LLB 1, Faculty of Law, University of Port Harcourt.',
    image: img
  }

  // Check if it's a material link: /materials/:courseSlug/:resourceSlug
  const m = path.match(/^\/materials\/([^/]+)\/([^/]+)$/)
  if (m) {
    const courseSlug = m[1]
    const resourceSlug = m[2]

    try {
      // Fetch the actual resource from Supabase
      const res = await fetch(`${SUPABASE_URL}/rest/v1/resources?course_slug=eq.${courseSlug}&slug=eq.${resourceSlug}&select=name,md,type`, {
        headers: {
          'apikey': SUPABASE_ANON_KEY,
          'Authorization': `Bearer ${SUPABASE_ANON_KEY}`
        }
      })
      const data = await res.json()
      
      if (data && data.length > 0) {
        const resource = data[0]
        // Clean up the markdown to create a short description
        let snippet = resource.md 
          ? resource.md.replace(/[#*`_]/g, '').replace(/\n+/g, ' ').substring(0, 150).trim() + '...'
          : 'Read this resource on the LLB 1 Hub.'
          
        return {
          title: `${resource.name} · ${courseSlug.toUpperCase()}`,
          description: snippet,
          image: img
        }
      }
    } catch (e) {
      console.error("Supabase fetch failed", e)
    }
  }

  return fallback
}
